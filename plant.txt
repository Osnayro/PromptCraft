window.sincronizarPlantillas = async function sincronizarPlantillas() {
    if (!window.db) {
        console.error('Firestore no disponible.');
        return;
    }

    if (!Array.isArray(window.plantillasSeed) || window.plantillasSeed.length === 0) {
        console.error('window.plantillasSeed no está definido. Vuelve a pegar el script completo antes de ejecutar sincronizarPlantillas().');
        return;
    }

    for (const plantilla of window.plantillasSeed) {
        const { id, ...datos } = plantilla;
        await window.db.collection('plantillas').doc(id).set(datos, { merge: true });
    }

    console.log(`✅ ${window.plantillasSeed.length} plantillas sincronizadas en Firestore.`);
};
